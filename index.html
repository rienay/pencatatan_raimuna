Integrasi Google Sheets

Sinkronisasikan pembukuan lokal Anda langsung ke Google Sheets secara berkala.

Masukkan URL Web App hasil deploy Apps Script pada Google Sheets Anda.

Sinkronisasi Otomatis Real-Time

Setiap transaksi baru, perubahan, atau penghapusan data akan otomatis terhubung secara langsung ke Google Sheets tanpa perlu proses manual.

Panduan Setup Google Sheets

Ikuti langkah-langkah berikut untuk mengaktifkan sinkronisasi otomatis:

  1. Buat spreadsheet baru di Google Sheets Anda.
  2. Buat tab lembar kerja dengan nama "Transaksi", "Kategori", dan "Sumber" (opsional).
  3. Pada menu bar atas, klik Ekstensi » Apps Script.
  4. Hapus kode bawaan, lalu salin dan tempelkan kode di bawah ini.
  5. Klik tombol Terapkan (Deploy) di bagian kanan atas » Penerapan baru (New deployment).
  6. Pilih jenis Aplikasi Web (Web App). Setel "Akses" ke **"Siapa saja" (Anyone)** agar website dapat mengaksesnya tanpa login.
  7. Klik Terapkan, berikan izin akses Google, lalu salin **URL Aplikasi Web** yang diberikan.
  8. Tempel URL tersebut ke kolom isian integrasi di sebelah kiri.
Kode Google Apps Script 
// ============================================================================
// KODE GOOGLE APPS SCRIPT - RAIMUNA CABANG CILACAP (VERSI MULTI-SHEET LENGKAP)
// Mendukung Semua Tab: Transaksi, Utang, Kwitansi, Sponsorship, Tenant, Jelantah
// Fitur: Auto Case-Insensitive, Upload Google Drive, Hapus Pintar, Bulk Sync
// ============================================================================

var FOLDER_PENGAMBILAN_UANG = '13yzUOz2qknJZ9fAHFmj9DeS3_p9riiID';
var FOLDER_NOTA = '1p7Tp1a_SyNlMaj95IfaLJ95z5bmezzGg';

// Definisi Header Standar untuk Tiap Tab
var HEADERS = {
  transaksi: ["id", "tanggal", "tipe", "kategori", "sumberDana", "pic", "nominal", "keterangan", "dateCreated", "bukti"],
  utang: ["id", "tanggal", "tipe", "nama", "sumberDana", "kategori", "nominal", "keterangan", "status", "paidTxId", "tanggalLunas", "dateCreated"],
  kwitansi: ["id", "tipeJenis", "no", "tgl", "dari", "nominal", "terbilang", "guna", "penerima", "nta", "dateCreated", "bukti"],
  sponsorship: ["id", "tipeJenis", "no", "tgl", "dari", "nominal", "terbilang", "guna", "penerima", "nta", "dateCreated", "bukti"],
  tenant: ["id", "tipeJenis", "no", "tgl", "dari", "nominal", "terbilang", "guna", "penerima", "nta", "dateCreated", "bukti"],
  jelantah: ["id", "tanggal", "kwarran", "kg", "nominal", "terbilang", "keterangan", "guna", "penerima", "dateCreated", "bukti"]
};

// ── FUNGSI PENCARIAN & PEMBUATAN TAB CASE-INSENSITIVE ────────────────────────
function getSheetCaseInsensitive(ss, name) {
  var sheets = ss.getSheets();
  var target = String(name).trim().toLowerCase();
  for (var i = 0; i < sheets.length; i++) {
    if (sheets[i].getName().trim().toLowerCase() === target) {
      return sheets[i];
    }
  }
  return null;
}

function getOrCreateSheet(ss, name, defaultHeaders) {
  var sheet = getSheetCaseInsensitive(ss, name);
  if (!sheet) {
    sheet = ss.insertSheet(name);
  }
  if (sheet.getLastRow() === 0 && defaultHeaders && defaultHeaders.length > 0) {
    sheet.appendRow(defaultHeaders);
  }
  return sheet;
}

// ── BACA DATA DARI SHEET MENJADI ARRAY OF OBJECTS ────────────────────────────
function readSheetData(sheet) {
  if (!sheet || sheet.getLastRow() <= 1) return [];
  var rows = sheet.getDataRange().getValues();
  if (rows.length < 2) return [];
  var headers = rows[0].map(function(h) { return String(h).trim(); });
  var result = [];
  
  for (var i = 1; i < rows.length; i++) {
    var row = rows[i];
    var isBlank = row.every(function(cell) { return cell === "" || cell === null || cell === undefined; });
    if (isBlank) continue;
    
    var obj = {};
    for (var j = 0; j < headers.length; j++) {
      if (headers[j]) {
        obj[headers[j]] = row[j];
      }
    }
    result.push(obj);
  }
  return result;
}

// ── AUTO MIGRATE KOLOM TRANSAKSI JIKA FORMAT LAMA ───────────────────────────
function autoMigrateColumns(sheet) {
  if (!sheet) return;
  var lastRow = sheet.getLastRow();
  if (lastRow === 0) return;
  var headers = sheet.getRange(1, 1, 1, sheet.getLastColumn()).getValues()[0];
  var catIndex = headers.indexOf("kategoriSumber");
  if (catIndex === -1) return;
  
  var values = sheet.getDataRange().getValues();
  sheet.insertColumnAfter(catIndex + 1);
  sheet.getRange(1, catIndex + 1).setValue("kategori");
  sheet.getRange(1, catIndex + 2).setValue("sumberDana");
  
  for (var i = 1; i < values.length; i++) {
    var tipe = values[i][2];
    var val = values[i][catIndex];
    var kategoriVal = (tipe === "OUT") ? (val || "-") : "-";
    var sumberVal = (tipe === "IN") ? (val || "-") : "-";
    sheet.getRange(i + 1, catIndex + 1).setValue(kategoriVal);
    sheet.getRange(i + 1, catIndex + 2).setValue(sumberVal);
  }
}

// ── UPLOAD LAMPIRAN KE GOOGLE DRIVE ─────────────────────────────────────────
function uploadAttachmentsToDrive(attachments, prefixId, folderId) {
  var urls = [];
  if (!attachments || !Array.isArray(attachments) || attachments.length === 0) {
    return urls;
  }
  try {
    var folder = DriveApp.getFolderById(folderId);
    for (var i = 0; i < attachments.length; i++) {
      var item = attachments[i];
      var base64Data = item.base64 || '';
      var contentType = item.type || 'image/jpeg';
      var fileName = (prefixId || 'lampiran') + '_' + (item.name || ('file_' + (i + 1)));

      if (base64Data.indexOf('base64,') !== -1) {
        base64Data = base64Data.split('base64,')[1];
      }
      if (!base64Data) continue;

      var blob = Utilities.newBlob(Utilities.base64Decode(base64Data), contentType, fileName);
      var file = folder.createFile(blob);
      file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
      urls.push(file.getUrl());
    }
  } catch (err) {
    Logger.log("Gagal upload Drive: " + err.toString());
  }
  return urls;
}

// ── UPSERT BARIS BY ID ──────────────────────────────────────────────────────
function upsertRowById(sheet, idValue, rowData, existingBuktiColIndex) {
  var idStr = String(idValue || '').trim();
  var foundRow = -1;
  var existingBukti = "";
  
  if (sheet.getLastRow() > 1) {
    var rows = sheet.getDataRange().getValues();
    for (var i = 1; i < rows.length; i++) {
      var rowId = String(rows[i][0] || '').trim();
      var rowNo = (rows[i].length > 2) ? String(rows[i][2] || '').trim() : '';
      if (idStr !== '' && (rowId === idStr || rowNo === idStr)) {
        foundRow = i + 1;
        if (existingBuktiColIndex !== undefined && rows[i][existingBuktiColIndex]) {
          existingBukti = rows[i][existingBuktiColIndex];
        }
        break;
      }
    }
  }
  
  if (foundRow !== -1) {
    sheet.getRange(foundRow, 1, 1, rowData.length).setValues([rowData]);
  } else {
    sheet.appendRow(rowData);
  }
  return existingBukti;
}

// ── DELETE BARIS BY ID / NO PINTAR ──────────────────────────────────────────
function deleteRowFromSheet(sheet, idValue, secondaryValue) {
  if (!sheet || sheet.getLastRow() <= 1) return false;
  var idStr = String(idValue || '').trim();
  var secStr = String(secondaryValue || '').trim();
  if (idStr === '' && secStr === '') return false;
  
  var rows = sheet.getDataRange().getValues();
  for (var i = 1; i < rows.length; i++) {
    var rowId = String(rows[i][0] || '').trim();
    var rowSec = (rows[i].length > 2) ? String(rows[i][2] || '').trim() : '';
    var rowContent = rows[i].join(" ");
    
    var match = false;
    if (idStr !== '' && (rowId === idStr || rowContent.indexOf(idStr) !== -1)) match = true;
    if (!match && secStr !== '' && (rowSec === secStr || rowContent.indexOf(secStr) !== -1)) match = true;
    
    if (match) {
      sheet.deleteRow(i + 1);
      return true;
    }
  }
  return false;
}

// doGet: Mengambil Semua Data dari Seluruh Sheet (Transaksi, Kwitansi, Utang,
// Sponsorship, Tenant, Jelantah)
// ============================================================================
function doGet(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  
  var sheetTx = getSheetCaseInsensitive(ss, "Transaksi");
  if (sheetTx) autoMigrateColumns(sheetTx);
  
  var sheetUt = getSheetCaseInsensitive(ss, "Utang");
  var sheetKw = getSheetCaseInsensitive(ss, "Kwitansi");
  var sheetSp = getSheetCaseInsensitive(ss, "Sponsorship");
  var sheetTenant = getSheetCaseInsensitive(ss, "Tenant") || getSheetCaseInsensitive(ss, "Stand");
  var sheetJlt = getSheetCaseInsensitive(ss, "Jelantah");

  // 1. Data Transaksi
  var txData = readSheetData(sheetTx);

  // 2. Data Utang
  var utData = readSheetData(sheetUt);

  // 3. Data Kwitansi Master & Tab Khusus
  var kwData = readSheetData(sheetKw);
  var spData = readSheetData(sheetSp);
  var tenantData = readSheetData(sheetTenant);
  var jltData = readSheetData(sheetJlt);

  // Gabungkan ke master kwitansi jika ada input langsung di sheet individual
  var kwIdMap = {};
  for (var k = 0; k < kwData.length; k++) {
    if (kwData[k].id) kwIdMap[String(kwData[k].id).trim()] = true;
  }

  // Tambahkan data dari tab Sponsorship jika belum ada di master kwitansi
  for (var s = 0; s < spData.length; s++) {
    var spItem = spData[s];
    var sId = String(spItem.id || ('SP-' + (s + 1))).trim();
    if (!kwIdMap[sId]) {
      if (!spItem.tipeJenis) spItem.tipeJenis = "SPONSORSHIP";
      kwData.push(spItem);
      kwIdMap[sId] = true;
    }
  }

  // Tambahkan data dari tab Tenant jika belum ada di master kwitansi
  for (var t = 0; t < tenantData.length; t++) {
    var tItem = tenantData[t];
    var tId = String(tItem.id || ('TNT-' + (t + 1))).trim();
    if (!kwIdMap[tId]) {
      if (!tItem.tipeJenis) tItem.tipeJenis = "STAND";
      kwData.push(tItem);
      kwIdMap[tId] = true;
    }
  }

  // Tambahkan data dari tab Jelantah jika belum ada di master kwitansi
  for (var j = 0; j < jltData.length; j++) {
    var jItem = jltData[j];
    var jId = String(jItem.id || ('JLT-' + (j + 1))).trim();
    if (!kwIdMap[jId]) {
      var jKw = {
        id: jId,
        tipeJenis: "JELANTAH",
        no: jItem.no || jId,
        tgl: jItem.tanggal || jItem.tgl || "",
        dari: jItem.kwarran || jItem.dari || "",
        nominal: jItem.nominal || (jItem.kg ? (jItem.kg + " KG") : "0 KG"),
        terbilang: jItem.terbilang || "",
        guna: jItem.guna || jItem.keterangan || "PERSYARATAN MINYAK JELANTAH",
        penerima: jItem.penerima || "Petugas Jelantah",
        nta: jItem.nta || "Ketua Panitia",
        dateCreated: jItem.dateCreated || "",
        bukti: jItem.bukti || ""
      };
      kwData.push(jKw);
      kwIdMap[jId] = true;
    }
  }

  return ContentService.createTextOutput(JSON.stringify({ 
    success: true, 
    data: txData,
    kwitansi: kwData,
    utang: utData,
    sponsorship: spData,
    tenant: tenantData,
    jelantah: jltData
  })).setMimeType(ContentService.MimeType.JSON);
}

// ============================================================================
// doPost: Menangani Sinkronisasi Data Masuk, Update, Hapus, & Upload
// ============================================================================
function doPost(e) {
  try {
    var payload = JSON.parse(e.postData.contents);
    var action = payload.action;
    var ss = SpreadsheetApp.getActiveSpreadsheet();

    // ── 0. TEST KONEKSI ───────────────────────────────────────────────────────
    if (action === "test") {
      return ContentService.createTextOutput(JSON.stringify({ 
        success: true, 
        message: "Koneksi Berhasil! Semua tab sheet aktif dan siap digunakan." 
      })).setMimeType(ContentService.MimeType.JSON);
    }

    // ── 1. TRANSAKSI (SYNC 1) ─────────────────────────────────────────────────
    if (action === "sync_single") {
      var sheetTx = getOrCreateSheet(ss, "Transaksi", HEADERS.transaksi);
      autoMigrateColumns(sheetTx);
      var tx = payload.transaction;
      
      var fileUrls = [];
      if (tx.attachment && Array.isArray(tx.attachment) && tx.attachment.length > 0) {
        var folderId = (tx.tipe === 'OUT') ? FOLDER_PENGAMBILAN_UANG : FOLDER_NOTA;
        fileUrls = uploadAttachmentsToDrive(tx.attachment, tx.id, folderId);
      }

      var existingBukti = "";
      var rows = sheetTx.getDataRange().getValues();
      var foundRow = -1;
      for (var i = 1; i < rows.length; i++) {
        if (String(rows[i][0]).trim() === String(tx.id).trim()) {
          foundRow = i + 1;
          existingBukti = rows[i][9] || "";
          break;
        }
      }

      var finalBukti = fileUrls.length > 0 ? fileUrls.join(', ') : existingBukti;
      var rowData = [
        tx.id, 
        tx.tanggal, 
        tx.tipe, 
        tx.kategori || "-", 
        tx.sumberDana || "-", 
        tx.pic, 
        tx.nominal, 
        tx.keterangan, 
        tx.dateCreated, 
        finalBukti
      ];

      if (foundRow !== -1) {
        sheetTx.getRange(foundRow, 1, 1, rowData.length).setValues([rowData]);
      } else {
        sheetTx.appendRow(rowData);
      }

      return ContentService.createTextOutput(JSON.stringify({ success: true, urls: fileUrls }))
        .setMimeType(ContentService.MimeType.JSON);
    }

    // ── 2. TRANSAKSI (HAPUS 1) ────────────────────────────────────────────────
    if (action === "delete_single") {
      var sheetTx = getSheetCaseInsensitive(ss, "Transaksi");
      deleteRowFromSheet(sheetTx, payload.id);
      return ContentService.createTextOutput(JSON.stringify({ success: true }))
        .setMimeType(ContentService.MimeType.JSON);
    }

    // ── 3. KWITANSI (SYNC 1) ──────────────────────────────────────────────────
    if (action === "sync_kwitansi") {
      var kw = payload.kwitansi;
      var fileUrls = [];
      if (kw.attachment && Array.isArray(kw.attachment) && kw.attachment.length > 0) {
        fileUrls = uploadAttachmentsToDrive(kw.attachment, kw.id || kw.no, FOLDER_NOTA);
      }

      var sheetKw = getOrCreateSheet(ss, "Kwitansi", HEADERS.kwitansi);
      var existingBukti = "";
      if (sheetKw.getLastRow() > 1) {
        var kRows = sheetKw.getDataRange().getValues();
        for (var k = 1; k < kRows.length; k++) {
          if (String(kRows[k][0]).trim() === String(kw.id).trim() || 
              String(kRows[k][2]).trim() === String(kw.no).trim()) {
            existingBukti = kRows[k][11] || "";
            break;
          }
        }
      }
      var finalBukti = fileUrls.length > 0 ? fileUrls.join(', ') : existingBukti;

      var kwRowData = [
        kw.id, 
        kw.tipeJenis || "SPONSORSHIP", 
        kw.no, 
        kw.tgl, 
        kw.dari, 
        kw.nominal, 
        kw.terbilang, 
        kw.guna, 
        kw.penerima, 
        kw.nta, 
        kw.dateCreated, 
        finalBukti
      ];
      upsertRowById(sheetKw, kw.id, kwRowData);

      var tipeUpper = String(kw.tipeJenis || '').toUpperCase();
      
      // Tab Sponsorship
      if (tipeUpper.indexOf('SPONSOR') !== -1 || (tipeUpper.indexOf('STAND') === -1 && tipeUpper.indexOf('JELANTAH') === -1 && tipeUpper.indexOf('TENANT') === -1)) {
        var sheetSp = getSheetCaseInsensitive(ss, "Sponsorship");
        if (sheetSp) upsertRowById(sheetSp, kw.id, kwRowData);
      }
      
      // Tab Tenant
      if (tipeUpper.indexOf('STAND') !== -1 || tipeUpper.indexOf('TENANT') !== -1) {
        var sheetTenant = getSheetCaseInsensitive(ss, "Tenant") || getSheetCaseInsensitive(ss, "Stand");
        if (sheetTenant) upsertRowById(sheetTenant, kw.id, kwRowData);
      }

      // Tab Jelantah
      if (tipeUpper.indexOf('JELANTAH') !== -1) {
        var sheetJlt = getSheetCaseInsensitive(ss, "Jelantah");
        if (sheetJlt) {
          var jltRow = [
            kw.id, 
            kw.tgl || "", 
            kw.dari || "", 
            kw.nominal || "", 
            kw.nominal || "", 
            kw.terbilang || "", 
            kw.guna || "", 
            kw.guna || "", 
            kw.penerima || "", 
            kw.dateCreated || "", 
            finalBukti
          ];
          upsertRowById(sheetJlt, kw.id, jltRow);
        }
      }

      return ContentService.createTextOutput(JSON.stringify({ success: true, urls: fileUrls }))
        .setMimeType(ContentService.MimeType.JSON);
    }

    // ── 4. KWITANSI (HAPUS 1) ─────────────────────────────────────────────────
    if (action === "delete_kwitansi") {
      var id = payload.id;
      var no = payload.no;
      deleteRowFromSheet(getSheetCaseInsensitive(ss, "Kwitansi"), id, no);
      deleteRowFromSheet(getSheetCaseInsensitive(ss, "Sponsorship"), id, no);
      deleteRowFromSheet(getSheetCaseInsensitive(ss, "Tenant") || getSheetCaseInsensitive(ss, "Stand"), id, no);
      deleteRowFromSheet(getSheetCaseInsensitive(ss, "Jelantah"), id, no);
      return ContentService.createTextOutput(JSON.stringify({ success: true }))
        .setMimeType(ContentService.MimeType.JSON);
    }

    // ── 5. RECAP JELANTAH (SYNC 1) ────────────────────────────────────────────
    if (action === "sync_jelantah") {
      var jlt = payload.jelantah;
      var fileUrls = [];
      if (jlt.attachment && Array.isArray(jlt.attachment) && jlt.attachment.length > 0) {
        fileUrls = uploadAttachmentsToDrive(jlt.attachment, jlt.id, FOLDER_NOTA);
      }

      var sheetJlt = getOrCreateSheet(ss, "Jelantah", HEADERS.jelantah);
      var existingBukti = "";
      if (sheetJlt.getLastRow() > 1) {
        var jRows = sheetJlt.getDataRange().getValues();
        for (var j = 1; j < jRows.length; j++) {
          if (String(jRows[j][0]).trim() === String(jlt.id).trim()) {
            existingBukti = jRows[j][10] || "";
            break;
          }
        }
      }
      var finalBukti = fileUrls.length > 0 ? fileUrls.join(', ') : existingBukti;

      var jltRowData = [
        jlt.id,
        jlt.tanggal || jlt.tgl,
        jlt.kwarran || jlt.dari || "-",
        jlt.kg || "",
        jlt.nominal || (jlt.kg ? (jlt.kg + " KG") : ""),
        jlt.terbilang || "",
        jlt.keterangan || "-",
        jlt.guna || "PENGAMBILAN MINYAK JELANTAH",
        jlt.penerima || "Petugas Jelantah",
        jlt.dateCreated,
        finalBukti
      ];
      upsertRowById(sheetJlt, jlt.id, jltRowData);

      var sheetKw = getSheetCaseInsensitive(ss, "Kwitansi");
      if (sheetKw) {
        var kwRow = [
          jlt.id,
          "JELANTAH",
          jlt.id,
          jlt.tanggal || jlt.tgl,
          jlt.kwarran || jlt.dari || "-",
          jlt.nominal || (jlt.kg ? (jlt.kg + " KG") : ""),
          jlt.terbilang || "",
          jlt.guna || "PERSYARATAN MINYAK JELANTAH",
          jlt.penerima || "Tri Soma Ananta Rahman",
          "Ketua Panitia",
          jlt.dateCreated,
          finalBukti
        ];
        upsertRowById(sheetKw, jlt.id, kwRow);
      }

      return ContentService.createTextOutput(JSON.stringify({ success: true, urls: fileUrls }))
        .setMimeType(ContentService.MimeType.JSON);
    }

    // ── 6. RECAP JELANTAH (HAPUS 1) ───────────────────────────────────────────
    if (action === "delete_jelantah") {
      deleteRowFromSheet(getSheetCaseInsensitive(ss, "Jelantah"), payload.id);
      deleteRowFromSheet(getSheetCaseInsensitive(ss, "Kwitansi"), payload.id);
      return ContentService.createTextOutput(JSON.stringify({ success: true }))
        .setMimeType(ContentService.MimeType.JSON);
    }

    // ── 7. UTANG / PIUTANG (SYNC 1) ───────────────────────────────────────────
    if (action === "sync_utang") {
      var utSheet = getOrCreateSheet(ss, "Utang", HEADERS.utang);
      var ut = payload.utang;
      var utRowData = [
        ut.id, 
        ut.tanggal, 
        ut.tipe, 
        ut.nama, 
        ut.sumberDana || "-", 
        ut.kategori || "-", 
        ut.nominal, 
        ut.keterangan, 
        ut.status, 
        ut.paidTxId || "-", 
        ut.tanggalLunas || "-", 
        ut.dateCreated
      ];
      upsertRowById(utSheet, ut.id, utRowData);
      return ContentService.createTextOutput(JSON.stringify({ success: true }))
        .setMimeType(ContentService.MimeType.JSON);
    }

    // ── 8. UTANG / PIUTANG (HAPUS 1) ──────────────────────────────────────────
    if (action === "delete_utang") {
      var utSheetDel = getSheetCaseInsensitive(ss, "Utang");
      deleteRowFromSheet(utSheetDel, payload.id);
      return ContentService.createTextOutput(JSON.stringify({ success: true }))
        .setMimeType(ContentService.MimeType.JSON);
    }

    // ── 9. PUSH BULK (Sinkronkan Seluruh Data ke Semua Tab Sekaligus) ──────────
    if (action === "push_bulk") {
      // 9.1 Transaksi
      var transactions = payload.transactions || [];
      var sheetTx = getOrCreateSheet(ss, "Transaksi", HEADERS.transaksi);
      if (sheetTx.getLastRow() > 1) {
        sheetTx.getRange(2, 1, sheetTx.getLastRow() - 1, sheetTx.getLastColumn()).clearContent();
      }
      if (transactions.length > 0) {
        var txRows = transactions.map(function(t) {
          return [t.id, t.tanggal, t.tipe, t.kategori || "-", t.sumberDana || "-", t.pic, t.nominal, t.keterangan, t.dateCreated, ""];
        });
        sheetTx.getRange(2, 1, txRows.length, txRows[0].length).setValues(txRows);
      }

      // 9.2 Utang
      var utangList = payload.utang || [];
      var sheetUt = getOrCreateSheet(ss, "Utang", HEADERS.utang);
      if (sheetUt.getLastRow() > 1) {
        sheetUt.getRange(2, 1, sheetUt.getLastRow() - 1, sheetUt.getLastColumn()).clearContent();
      }
      if (utangList.length > 0) {
        var utRows = utangList.map(function(u) {
          return [u.id, u.tanggal, u.tipe, u.nama, u.sumberDana || "-", u.kategori || "-", u.nominal, u.keterangan, u.status, u.paidTxId || "-", u.tanggalLunas || "-", u.dateCreated];
        });
        sheetUt.getRange(2, 1, utRows.length, utRows[0].length).setValues(utRows);
      }

      // 9.3 Kwitansi Master
      var kwitansiList = payload.kwitansi || [];
      var sheetKw = getOrCreateSheet(ss, "Kwitansi", HEADERS.kwitansi);
      if (sheetKw.getLastRow() > 1) {
        sheetKw.getRange(2, 1, sheetKw.getLastRow() - 1, sheetKw.getLastColumn()).clearContent();
      }
      if (kwitansiList.length > 0) {
        var kwRows = kwitansiList.map(function(k) {
          return [k.id, k.tipeJenis || "SPONSORSHIP", k.no, k.tgl, k.dari, k.nominal, k.terbilang, k.guna, k.penerima, k.nta, k.dateCreated, ""];
        });
        sheetKw.getRange(2, 1, kwRows.length, kwRows[0].length).setValues(kwRows);
      }

      // 9.4 Tab Sponsorship Khusus
      var sheetSp = getSheetCaseInsensitive(ss, "Sponsorship");
      if (sheetSp) {
        if (sheetSp.getLastRow() === 0) sheetSp.appendRow(HEADERS.sponsorship);
        else if (sheetSp.getLastRow() > 1) sheetSp.getRange(2, 1, sheetSp.getLastRow() - 1, sheetSp.getLastColumn()).clearContent();
        
        var spItems = kwitansiList.filter(function(k) {
          var tp = String(k.tipeJenis || '').toUpperCase();
          return tp.indexOf('SPONSOR') !== -1 || (tp.indexOf('STAND') === -1 && tp.indexOf('TENANT') === -1 && tp.indexOf('JELANTAH') === -1);
        });
        if (spItems.length > 0) {
          var spRows = spItems.map(function(k) {
            return [k.id, k.tipeJenis || "SPONSORSHIP", k.no, k.tgl, k.dari, k.nominal, k.terbilang, k.guna, k.penerima, k.nta, k.dateCreated, ""];
          });
          sheetSp.getRange(2, 1, spRows.length, spRows[0].length).setValues(spRows);
        }
      }

      // 9.5 Tab Tenant Khusus
      var sheetTenant = getSheetCaseInsensitive(ss, "Tenant") || getSheetCaseInsensitive(ss, "Stand");
      if (sheetTenant) {
        if (sheetTenant.getLastRow() === 0) sheetTenant.appendRow(HEADERS.tenant);
        else if (sheetTenant.getLastRow() > 1) sheetTenant.getRange(2, 1, sheetTenant.getLastRow() - 1, sheetTenant.getLastColumn()).clearContent();
        
        var tenantItems = kwitansiList.filter(function(k) {
          var tp = String(k.tipeJenis || '').toUpperCase();
          return tp.indexOf('STAND') !== -1 || tp.indexOf('TENANT') !== -1;
        });
        if (tenantItems.length > 0) {
          var tntRows = tenantItems.map(function(k) {
            return [k.id, k.tipeJenis || "STAND", k.no, k.tgl, k.dari, k.nominal, k.terbilang, k.guna, k.penerima, k.nta, k.dateCreated, ""];
          });
          sheetTenant.getRange(2, 1, tntRows.length, tntRows[0].length).setValues(tntRows);
        }
      }

      // 9.6 Tab Jelantah Khusus
      var sheetJlt = getSheetCaseInsensitive(ss, "Jelantah");
      if (sheetJlt) {
        if (sheetJlt.getLastRow() === 0) sheetJlt.appendRow(HEADERS.jelantah);
        else if (sheetJlt.getLastRow() > 1) sheetJlt.getRange(2, 1, sheetJlt.getLastRow() - 1, sheetJlt.getLastColumn()).clearContent();
        
        var jelantahList = payload.jelantah || [];
        if (jelantahList.length === 0) {
          jelantahList = kwitansiList.filter(function(k) {
            return String(k.tipeJenis || '').toUpperCase().indexOf('JELANTAH') !== -1;
          }).map(function(k) {
            return {
              id: k.id,
              tanggal: k.tgl,
              kwarran: k.dari,
              kg: k.nominal,
              nominal: k.nominal,
              terbilang: k.terbilang,
              keterangan: k.guna,
              guna: k.guna,
              penerima: k.penerima,
              dateCreated: k.dateCreated
            };
          });
        }
        
        if (jelantahList.length > 0) {
          var jltRows = jelantahList.map(function(j) {
            return [
              j.id, 
              j.tanggal || j.tgl, 
              j.kwarran || j.dari || "-", 
              j.kg || "", 
              j.nominal || (j.kg ? (j.kg + " KG") : ""), 
              j.terbilang || "", 
              j.keterangan || "-", 
              j.guna || "PENGAMBILAN MINYAK JELANTAH", 
              j.penerima || "Petugas Jelantah", 
              j.dateCreated, 
              ""
            ];
          });
          sheetJlt.getRange(2, 1, jltRows.length, jltRows[0].length).setValues(jltRows);
        }
      }

      return ContentService.createTextOutput(JSON.stringify({ 
        success: true, 
        count: transactions.length,
        message: "Berhasil mengekspor semua data ke seluruh tab sheet!"
      })).setMimeType(ContentService.MimeType.JSON);
    }

    return ContentService.createTextOutput(JSON.stringify({ 
      success: false, 
      error: "Action tidak dikenal: " + action 
    })).setMimeType(ContentService.MimeType.JSON);

  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({ 
      success: false, 
      error: err.toString() 
    })).setMimeType(ContentService.MimeType.JSON);
  }
}
