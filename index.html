Integrasi Google Sheets

Sinkronisasikan pembukuan lokal Anda langsung ke Google Sheets secara berkala.

Masukkan URL Web App hasil deploy Apps Script pada Google Sheets Anda.

Sinkronisasi Otomatis Real-Time

Setiap transaksi baru, perubahan, atau penghapusan data akan otomatis terhubung secara langsung ke Google Sheets tanpa perlu proses manual.

Panduan Setup Google Sheets

Ikuti langkah-langkah berikut untuk mengaktifkan sinkronisasi otomatis:

  1. Buat spreadsheet baru di Google Sheets Anda.
  2. Buat tab lembar kerja dengan nama "Transaksi", "Kategori", dan "Sumber" (opsional).
  3. Pada menu bar atas, klik Ekstensi » Apps Script.
  4. Hapus kode bawaan, lalu salin dan tempelkan kode di bawah ini.
  5. Klik tombol Terapkan (Deploy) di bagian kanan atas » Penerapan baru (New deployment).
  6. Pilih jenis Aplikasi Web (Web App). Setel "Akses" ke **"Siapa saja" (Anyone)** agar website dapat mengaksesnya tanpa login.
  7. Klik Terapkan, berikan izin akses Google, lalu salin **URL Aplikasi Web** yang diberikan.
  8. Tempel URL tersebut ke kolom isian integrasi di sebelah kiri.
Kode Google Apps Script 
// KODE APPS SCRIPT UNTUK GOOGLE SHEETS (SINKRONISASI 3 TAB: Transaksi, Kwitansi, Utang)
function doGet(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheetTx = ss.getSheetByName("Transaksi");
  var sheetKw = ss.getSheetByName("Kwitansi");
  var sheetUt = ss.getSheetByName("Utang");
  
  // 1. Data Transaksi
  var txData = [];
  if (sheetTx && sheetTx.getLastRow() > 1) {
    autoMigrateColumns(sheetTx);
    var rows = sheetTx.getDataRange().getValues();
    var headers = rows[0];
    for (var i = 1; i < rows.length; i++) {
      var row = rows[i];
      var obj = {};
      for (var j = 0; j < headers.length; j++) {
        obj[headers[j]] = row[j];
      }
      txData.push(obj);
    }
  }

  // 2. Data Kwitansi / Tanda Terima Jelantah
  var kwData = [];
  if (sheetKw && sheetKw.getLastRow() > 1) {
    var rowsKw = sheetKw.getDataRange().getValues();
    var headersKw = rowsKw[0];
    for (var k = 1; k < rowsKw.length; k++) {
      var rowKw = rowsKw[k];
      var objKw = {};
      for (var m = 0; m < headersKw.length; m++) {
        objKw[headersKw[m]] = rowKw[m];
      }
      kwData.push(objKw);
    }
  }

  // 3. Data Utang / Piutang
  var utData = [];
  if (sheetUt && sheetUt.getLastRow() > 1) {
    var rowsUt = sheetUt.getDataRange().getValues();
    var headersUt = rowsUt[0];
    for (var u = 1; u < rowsUt.length; u++) {
      var rowUt = rowsUt[u];
      var objUt = {};
      for (var v = 0; v < headersUt.length; v++) {
        objUt[headersUt[v]] = rowUt[v];
      }
      utData.push(objUt);
    }
  }
  
  return ContentService.createTextOutput(JSON.stringify({ 
    success: true, 
    data: txData,
    kwitansi: kwData,
    utang: utData
  })).setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  try {
    var payload = JSON.parse(e.postData.contents);
    var action = payload.action;
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    
    var sheet = ss.getSheetByName("Transaksi") || ss.insertSheet("Transaksi");
    
    if (sheet.getLastRow() === 0) {
      sheet.appendRow(["id", "tanggal", "tipe", "kategori", "sumberDana", "pic", "nominal", "keterangan", "dateCreated"]);
    } else {
      autoMigrateColumns(sheet);
    }
    
    if (action === "test") {
      return ContentService.createTextOutput(JSON.stringify({ success: true, message: "Koneksi Berhasil!" }))
        .setMimeType(ContentService.MimeType.JSON);
    }
    
    // Sync 1 Transaksi Keuangan
    if (action === "sync_single") {
      var tx = payload.transaction;
      var rows = sheet.getDataRange().getValues();
      var foundRow = -1;
      for (var i = 1; i < rows.length; i++) {
        if (rows[i][0] === tx.id) {
          foundRow = i + 1;
          break;
        }
      }
      var rowData = [
        tx.id, tx.tanggal, tx.tipe, tx.kategori || "-", tx.sumberDana || "-", tx.pic, tx.nominal, tx.keterangan, tx.dateCreated
      ];
      if (foundRow !== -1) {
        sheet.getRange(foundRow, 1, 1, rowData.length).setValues([rowData]);
      } else {
        sheet.appendRow(rowData);
      }
      return ContentService.createTextOutput(JSON.stringify({ success: true }))
        .setMimeType(ContentService.MimeType.JSON);
    }

    // Push Bulk Transaksi, Kwitansi, & Utang
    if (action === "push_bulk") {
      var transactions = payload.transactions || [];
      if (sheet.getLastRow() > 1) {
        sheet.getRange(2, 1, sheet.getLastRow() - 1, sheet.getLastColumn()).clearContent();
      }
      var rowsToAppend = transactions.map(function(tx) {
        return [
          tx.id, tx.tanggal, tx.tipe, tx.kategori || "-", tx.sumberDana || "-", tx.pic, tx.nominal, tx.keterangan, tx.dateCreated
        ];
      });
      if (rowsToAppend.length > 0) {
        sheet.getRange(2, 1, rowsToAppend.length, rowsToAppend[0].length).setValues(rowsToAppend);
      }

      // Kwitansi Bulk
      var kwitansiList = payload.kwitansi || [];
      if (kwitansiList.length > 0) {
        var kwSheet = ss.getSheetByName("Kwitansi") || ss.insertSheet("Kwitansi");
        if (kwSheet.getLastRow() === 0) {
          kwSheet.appendRow(["id", "tipeJenis", "no", "tgl", "dari", "nominal", "terbilang", "guna", "penerima", "nta", "dateCreated"]);
        } else if (kwSheet.getLastRow() > 1) {
          kwSheet.getRange(2, 1, kwSheet.getLastRow() - 1, kwSheet.getLastColumn()).clearContent();
        }
        var kwRowsToAppend = kwitansiList.map(function(kw) {
          return [
            kw.id, kw.tipeJenis, kw.no, kw.tgl, kw.dari, kw.nominal, kw.terbilang, kw.guna, kw.penerima, kw.nta, kw.dateCreated
          ];
        });
        if (kwRowsToAppend.length > 0) {
          kwSheet.getRange(2, 1, kwRowsToAppend.length, kwRowsToAppend[0].length).setValues(kwRowsToAppend);
        }
      }

      // Utang Bulk
      var utangList = payload.utang || [];
      if (utangList.length > 0) {
        var utSheet = ss.getSheetByName("Utang") || ss.insertSheet("Utang");
        if (utSheet.getLastRow() === 0) {
          utSheet.appendRow(["id", "tanggal", "tipe", "nama", "sumberDana", "kategori", "nominal", "keterangan", "status", "paidTxId", "tanggalLunas", "dateCreated"]);
        } else if (utSheet.getLastRow() > 1) {
          utSheet.getRange(2, 1, utSheet.getLastRow() - 1, utSheet.getLastColumn()).clearContent();
        }
        var utRowsToAppend = utangList.map(function(ut) {
          return [
            ut.id, ut.tanggal, ut.tipe, ut.nama, ut.sumberDana || "-", ut.kategori || "-", ut.nominal, ut.keterangan, ut.status, ut.paidTxId || "-", ut.tanggalLunas || "-", ut.dateCreated
          ];
        });
        if (utRowsToAppend.length > 0) {
          utSheet.getRange(2, 1, utRowsToAppend.length, utRowsToAppend[0].length).setValues(utRowsToAppend);
        }
      }

      return ContentService.createTextOutput(JSON.stringify({ success: true, count: rowsToAppend.length }))
        .setMimeType(ContentService.MimeType.JSON);
    }
    
    // Hapus 1 Transaksi
    if (action === "delete_single") {
      var txId = String(payload.id).trim();
      var rows = sheet.getDataRange().getValues();
      for (var i = 1; i < rows.length; i++) {
        var rowStr = rows[i].join(" ");
        if (rows[i][0] == txId || rowStr.indexOf(txId) !== -1) {
          sheet.deleteRow(i + 1);
          break;
        }
      }
      return ContentService.createTextOutput(JSON.stringify({ success: true }))
        .setMimeType(ContentService.MimeType.JSON);
    }
    
    // Sync Kwitansi
    if (action === "sync_kwitansi") {
      var kwSheet = ss.getSheetByName("Kwitansi") || ss.insertSheet("Kwitansi");
      if (kwSheet.getLastRow() === 0) {
        kwSheet.appendRow(["id", "tipeJenis", "no", "tgl", "dari", "nominal", "terbilang", "guna", "penerima", "nta", "dateCreated"]);
      }
      var kw = payload.kwitansi;
      var kwRows = kwSheet.getDataRange().getValues();
      var foundKwRow = -1;
      for (var k = 1; k < kwRows.length; k++) {
        if (kwRows[k][0] === kw.id) {
          foundKwRow = k + 1;
          break;
        }
      }
      var kwRowData = [
        kw.id, kw.tipeJenis, kw.no, kw.tgl, kw.dari, kw.nominal, kw.terbilang, kw.guna, kw.penerima, kw.nta, kw.dateCreated
      ];
      if (foundKwRow !== -1) {
        kwSheet.getRange(foundKwRow, 1, 1, kwRowData.length).setValues([kwRowData]);
      } else {
        kwSheet.appendRow(kwRowData);
      }
      return ContentService.createTextOutput(JSON.stringify({ success: true }))
        .setMimeType(ContentService.MimeType.JSON);
    }

    // Hapus 1 Kwitansi
    if (action === "delete_kwitansi") {
      var kwSheetDel = ss.getSheetByName("Kwitansi");
      if (kwSheetDel) {
        var kwId = String(payload.id || '').trim();
        var kwNo = String(payload.no || '').trim();
        var cleanKwNo = kwNo.replace(/^No\.\s*/i, '').trim();
        var kwRowsDel = kwSheetDel.getDataRange().getValues();
        for (var kd = 1; kd < kwRowsDel.length; kd++) {
          var rowDataStr = kwRowsDel[kd].join(" ");
          var colA = String(kwRowsDel[kd][0]).trim();
          var colB = String(kwRowsDel[kd][1]).trim();
          var cleanColB = colB.replace(/^No\.\s*/i, '').trim();
          
          if ((kwId && (colA === kwId || rowDataStr.indexOf(kwId) !== -1)) ||
              (cleanKwNo && (cleanColB === cleanKwNo || colB.indexOf(cleanKwNo) !== -1))) {
            kwSheetDel.deleteRow(kd + 1);
            break;
          }
        }
      }
      return ContentService.createTextOutput(JSON.stringify({ success: true }))
        .setMimeType(ContentService.MimeType.JSON);
    }

    // Sync Utang / Piutang
    if (action === "sync_utang") {
      var utSheet = ss.getSheetByName("Utang") || ss.insertSheet("Utang");
      if (utSheet.getLastRow() === 0) {
        utSheet.appendRow(["id", "tanggal", "tipe", "nama", "sumberDana", "kategori", "nominal", "keterangan", "status", "paidTxId", "tanggalLunas", "dateCreated"]);
      }
      var ut = payload.utang;
      var utRows = utSheet.getDataRange().getValues();
      var foundUtRow = -1;
      for (var u = 1; u < utRows.length; u++) {
        if (utRows[u][0] === ut.id) {
          foundUtRow = u + 1;
          break;
        }
      }
      var utRowData = [
        ut.id, ut.tanggal, ut.tipe, ut.nama, ut.sumberDana || "-", ut.kategori || "-", ut.nominal, ut.keterangan, ut.status, ut.paidTxId || "-", ut.tanggalLunas || "-", ut.dateCreated
      ];
      if (foundUtRow !== -1) {
        utSheet.getRange(foundUtRow, 1, 1, utRowData.length).setValues([utRowData]);
      } else {
        utSheet.appendRow(utRowData);
      }
      return ContentService.createTextOutput(JSON.stringify({ success: true }))
        .setMimeType(ContentService.MimeType.JSON);
    }

    // Hapus 1 Utang
    if (action === "delete_utang") {
      var utSheetDel = ss.getSheetByName("Utang");
      if (utSheetDel) {
        var utId = String(payload.id || '').trim();
        var utRowsDel = utSheetDel.getDataRange().getValues();
        for (var d = 1; d < utRowsDel.length; d++) {
          var utRowStr = utRowsDel[d].join(" ");
          if (utRowsDel[d][0] == utId || utRowsDel[d][1] == utId || utRowStr.indexOf(utId) !== -1) {
            utSheetDel.deleteRow(d + 1);
            break;
          }
        }
      }
      }
      return ContentService.createTextOutput(JSON.stringify({ success: true }))
        .setMimeType(ContentService.MimeType.JSON);
    }
    
    return ContentService.createTextOutput(JSON.stringify({ success: false, error: "Action tidak dikenal" }))
      .setMimeType(ContentService.MimeType.JSON);
      
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({ success: false, error: err.toString() }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}

/**
 * Otomatis memisahkan kolom 'kategoriSumber' di Google Sheet menjadi 'kategori' & 'sumberDana'
 */
function autoMigrateColumns(sheet) {
  var lastRow = sheet.getLastRow();
  if (lastRow === 0) return;
  
  var range = sheet.getRange(1, 1, lastRow, sheet.getLastColumn());
  var values = range.getValues();
  var headers = values[0];
  
  var catIndex = headers.indexOf("kategoriSumber");
  if (catIndex === -1) return; // Sudah versi baru
  
  sheet.insertColumnAfter(catIndex + 1);
  sheet.getRange(1, catIndex + 1).setValue("kategori");
  sheet.getRange(1, catIndex + 2).setValue("sumberDana");
  
  for (var i = 1; i < values.length; i++) {
    var tipe = values[i][2];
    var val = values[i][catIndex];
    
    var kategoriVal = "-";
    var sumberVal = "-";
    
    if (tipe === "IN") {
      kategoriVal = "-";
      sumberVal = val || "-";
    } else {
      kategoriVal = val || "-";
      sumberVal = "-";
    }
    
    sheet.getRange(i + 1, catIndex + 1).setValue(kategoriVal);
    sheet.getRange(i + 1, catIndex + 2).setValue(sumberVal);
  }
}
