Integrasi Google Sheets

Sinkronisasikan pembukuan lokal Anda langsung ke Google Sheets secara berkala.

Masukkan URL Web App hasil deploy Apps Script pada Google Sheets Anda.

Sinkronisasi Massal

Kirim semua data lokal ke Google Sheet atau unduh ulang data yang tersimpan di Google Sheet.

Panduan Setup Google Sheets

Ikuti langkah-langkah berikut untuk mengaktifkan sinkronisasi otomatis:

  1. Buat spreadsheet baru di Google Sheets Anda.
  2. Buat tab lembar kerja dengan nama "Transaksi", "Kategori", dan "Sumber" (opsional).
  3. Pada menu bar atas, klik Ekstensi » Apps Script.
  4. Hapus kode bawaan, lalu salin dan tempelkan kode di bawah ini.
  5. Klik tombol Terapkan (Deploy) di bagian kanan atas » Penerapan baru (New deployment).
  6. Pilih jenis Aplikasi Web (Web App). Setel "Akses" ke **"Siapa saja" (Anyone)** agar website dapat mengaksesnya tanpa login.
  7. Klik Terapkan, berikan izin akses Google, lalu salin **URL Aplikasi Web** yang diberikan.
  8. Tempel URL tersebut ke kolom isian integrasi di sebelah kiri.
Kode Google Apps Script 
// KODE APPS SCRIPT UNTUK GOOGLE SHEETS
function doGet(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("Transaksi");
  if (!sheet) {
    return ContentService.createTextOutput(JSON.stringify({ success: false, error: "Sheet Transaksi tidak ditemukan" })).setMimeType(ContentService.MimeType.JSON);
  }
  var data = [];
  var rows = sheet.getDataRange().getValues();
  var headers = rows[0];
  for (var i = 1; i < rows.length; i++) {
    var row = rows[i];
    var obj = {};
    for (var j = 0; j < headers.length; j++) {
      obj[headers[j]] = row[j];
    }
    data.push(obj);
  }
  return ContentService.createTextOutput(JSON.stringify({ success: true, data: data }))
    .setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  try {
    var payload = JSON.parse(e.postData.contents);
    var action = payload.action;
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    
    // Pastikan sheet Transaksi ada
    var sheet = ss.getSheetByName("Transaksi") || ss.insertSheet("Transaksi");
    
    if (sheet.getLastRow() === 0) {
      // Inisialisasi header jika kosong
      sheet.appendRow(["id", "tanggal", "tipe", "kategoriSumber", "pic", "nominal", "keterangan", "dateCreated"]);
    }
    
    if (action === "test") {
      return ContentService.createTextOutput(JSON.stringify({ success: true, message: "Koneksi Berhasil!" }))
        .setMimeType(ContentService.MimeType.JSON);
    }
    
    if (action === "push_bulk") {
      // Hapus baris lama kecuali header
      var lastRow = sheet.getLastRow();
      if (lastRow > 1) {
        sheet.deleteRows(2, lastRow - 1);
      }
      var txs = payload.transactions;
      for (var i = 0; i < txs.length; i++) {
        var tx = txs[i];
        sheet.appendRow([
          tx.id,
          tx.tanggal,
          tx.tipe,
          tx.kategoriSumber,
          tx.pic,
          tx.nominal,
          tx.keterangan,
          tx.dateCreated
        ]);
      }
      return ContentService.createTextOutput(JSON.stringify({ success: true, count: txs.length }))
        .setMimeType(ContentService.MimeType.JSON);
    }
    
    if (action === "sync_single") {
      var tx = payload.transaction;
      // Periksa apakah ID transaksi sudah ada untuk di-update, atau tambahkan baris baru
      var rows = sheet.getDataRange().getValues();
      var foundRow = -1;
      for (var i = 1; i < rows.length; i++) {
        if (rows[i][0] === tx.id) {
          foundRow = i + 1;
          break;
        }
      }
      
      var rowData = [
        tx.id,
        tx.tanggal,
        tx.tipe,
        tx.kategoriSumber,
        tx.pic,
        tx.nominal,
        tx.keterangan,
        tx.dateCreated
      ];
      
      if (foundRow !== -1) {
        // Update baris yang ditemukan
        sheet.getRange(foundRow, 1, 1, rowData.length).setValues([rowData]);
      } else {
        // Tambahkan baris baru
        sheet.appendRow(rowData);
      }
      return ContentService.createTextOutput(JSON.stringify({ success: true }))
        .setMimeType(ContentService.MimeType.JSON);
    }
    
    if (action === "delete_single") {
      var txId = payload.id;
      var rows = sheet.getDataRange().getValues();
      for (var i = 1; i < rows.length; i++) {
        if (rows[i][0] === txId) {
          sheet.deleteRow(i + 1);
          break;
        }
      }
      return ContentService.createTextOutput(JSON.stringify({ success: true }))
        .setMimeType(ContentService.MimeType.JSON);
    }
    
    return ContentService.createTextOutput(JSON.stringify({ success: false, error: "Action tidak dikenal" }))
      .setMimeType(ContentService.MimeType.JSON);
      
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({ success: false, error: err.toString() }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}
